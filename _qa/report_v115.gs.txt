/* ============================================================
 * v115: 날짜별·세트별 현황을 '진도DB' 탭 기준으로 집계
 *  - 지금 앱(index/writing/reading)은 진도 기록을 '진도DB' 탭에만 저장합니다.
 *  - 그래서 기존 '기록' 탭(2026-11-10 이전 자료)과 진도DB를 함께 병합해 보여줍니다.
 *  - 인증: 기존과 똑같이 선생님 비밀번호를 매번 확인합니다(admin_).
 * ============================================================ */

/* 날짜를 'yyyy-MM-dd'로 통일 (Date/문자열 모두) */
function repV115Day_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
  var t = pdText_(v).trim().replace(/[.\/]/g, '-');
  if (/^\d{4}-\d{1,2}-\d{1,2}$/.test(t)) {
    var p = t.split('-');
    return p[0] + '-' + ('0' + p[1]).slice(-2) + '-' + ('0' + p[2]).slice(-2);
  }
  if (/^\d{4}-\d{2}-\d{2}[T ]/.test(t)) return t.slice(0, 10);
  return t;
}

/* 진도DB 전체 행 (없거나 오류면 빈 배열) */
function repV115DbRows_() {
  try {
    var rows = pdRows_() || [];
    return Array.isArray(rows) ? rows : [];
  } catch (e) {
    return [];
  }
}

/* 점수를 남기는 학습 모드인지 (진도DB는 '테스트'/'문제'/'문제풀이') */
function repV115ScoreMode_(mode) {
  var m = pdText_(mode).trim();
  return m === '테스트' || m === '문제' || m === '문제풀이';
}

function repV115DaySlot_(by, ban, name, type, set) {
  var k = ban + '|' + name;
  if (!by[k]) by[k] = {};
  var sk = type + '|' + set;
  if (!by[k][sk]) by[k][sk] = { type: type, set: set, modes: [], best: null };
  return by[k][sk];
}

/* 날짜별 현황 — '기록' 탭 + 진도DB 병합 */
function adminDayReportV115(pw, ban, day) {
  admin_(pw);
  ban = s_(ban);
  day = repV115Day_(day);
  var studs = studentsOf_(ban);
  var by = {};

  /* 1) 옛 기록 탭 (2026-11-10 이전 데이터) */
  rows_(SH.LOG).forEach(function (r) {
    var t = timeStr_(r[0]);
    if (t.slice(0, 10) !== day) return;
    if (ban && s_(r[1]) !== ban) return;
    var e = repV115DaySlot_(by, s_(r[1]), s_(r[2]), s_(r[3]), s_(r[4]));
    var mode = s_(r[5]);
    if (e.modes.indexOf(mode) < 0) e.modes.push(mode);
    if (mode === '테스트' || mode === '문제풀이') {
      var sc0 = Number(r[6]) || 0;
      if (!e.best || sc0 > e.best.score) {
        e.best = { score: sc0, correct: Number(r[7]) || 0, total: Number(r[8]) || 0, wrong: s_(r[9]) };
      }
    }
  });

  /* 2) 진도DB 탭 (지금 앱이 실제로 저장하는 곳) */
  repV115DbRows_().forEach(function (r) {
    if (!r) return;
    if (repV115Day_(r.d) !== day) return;
    var b = pdText_(r.b).trim(), n = pdText_(r.n).trim(), st = pdText_(r.s).trim(), m = pdText_(r.m).trim();
    if (!b || !n || !st || !m) return;
    if (ban && b !== ban) return;
    var e = repV115DaySlot_(by, b, n, '단어', st);
    if (e.modes.indexOf(m) < 0) e.modes.push(m);
    var sc = Number(r.sc) || 0;
    if (repV115ScoreMode_(m)) {
      if (!e.best || sc > e.best.score) {
        e.best = { score: sc, correct: null, total: null, wrong: '', at: pdText_(r.u) };
      }
    }
  });

  var md = Number(day.slice(5, 7)) + '/' + Number(day.slice(8, 10));
  var out = studs.map(function (st) {
    var row = by[st.ban + '|' + st.name];
    var items = row ? Object.keys(row).map(function (k) { return row[k]; }) : [];
    var nick = josa_(st.name);
    var msg;
    if (!items.length) {
      msg = '안녕하세요.탑에듀학원입니다^^ ' + md + ' ' + nick + '는 아직 영어 온라인 학습을 하지 않았습니다. 가정에서도 한 번 챙겨봐 주세요. 감사합니다♡';
    } else {
      var lines = items.map(function (e) {
        if (e.best) {
          if (e.best.correct === null || e.best.correct === undefined) {
            return '· ' + e.set + ' ' + e.best.score + '점 (' + e.modes.join(', ') + ')';
          }
          var line = '· ' + e.set + (e.type === '단어' ? ' 테스트 ' : ' ') + e.best.correct + '/' + e.best.total;
          if (e.best.wrong && e.type === '단어') line += ' (틀린 단어: ' + e.best.wrong + ')';
          return line;
        }
        return '· ' + e.set + ' ' + e.modes.join(', ') + ' 학습';
      });
      msg = '안녕하세요.탑에듀학원입니다^^ ' + md + ' ' + nick + '의 영어 온라인 학습 결과입니다.\n' + lines.join('\n') + '\n감사합니다♡';
    }
    return { ban: st.ban, name: st.name, did: items.length > 0, items: items, msg: msg };
  });
  return { day: day, students: out };
}

/* 세트별 현황 — '기록' 탭 + 진도DB 병합 */
function adminSetReportV115(pw, ban, type, setName) {
  admin_(pw);
  ban = s_(ban);
  type = s_(type) || '단어';
  setName = s_(setName);

  var by = {}, miss = {};
  function slot(k) {
    if (!by[k]) by[k] = { modes: {}, best: null, last: '', count: 0 };
    return by[k];
  }

  /* 1) 옛 기록 탭 */
  rows_(SH.LOG).forEach(function (r) {
    if (s_(r[3]) !== type || s_(r[4]) !== setName) return;
    if (ban && s_(r[1]) !== ban) return;
    var o = slot(s_(r[1]) + '|' + s_(r[2]));
    var mode = s_(r[5]), sc = Number(r[6]) || 0, t = timeStr_(r[0]);
    o.modes[mode] = (o.modes[mode] || 0) + 1;
    o.count++;
    if (t > o.last) o.last = t;
    if (mode === '테스트' || mode === '문제풀이') {
      if (o.best === null || sc > o.best.score) {
        o.best = { score: sc, correct: Number(r[7]) || 0, total: Number(r[8]) || 0, wrong: s_(r[9]) };
      }
      s_(r[9]).split(/,\s*/).filter(String).forEach(function (x) {
        x = x.replace(/\(서술형 확인\)/, '');
        miss[x] = (miss[x] || 0) + 1;
      });
    }
  });

  /* 2) 진도DB 탭 */
  repV115DbRows_().forEach(function (r) {
    if (!r) return;
    var b = pdText_(r.b).trim(), n = pdText_(r.n).trim(), st = pdText_(r.s).trim(), mode = pdText_(r.m).trim();
    if (!b || !n || !st || !mode) return;
    if (st !== setName) return;
    if (ban && b !== ban) return;
    var sc = Number(r.sc) || 0, t = pdText_(r.u).trim();
    var o = slot(b + '|' + n);
    o.modes[mode] = (o.modes[mode] || 0) + 1;
    o.count++;
    if (t > o.last) o.last = t;
    if (repV115ScoreMode_(mode)) {
      if (o.best === null || sc > o.best.score) {
        o.best = { score: sc, correct: null, total: null, wrong: '' };
      }
    }
  });

  /* 대상: 이 세트가 배정된 학생 + 실제 기록이 있는 학생 */
  var list = type === '단어' ? rows_(SH.WORD) : rows_(SH.PROB);
  var studs = studentsOf_(ban).filter(function (st) {
    if (by[st.ban + '|' + st.name]) return true;
    return list.some(function (r) { return s_(r[0]) === setName && forClass_(r[1], st.ban); });
  });

  var out = studs.map(function (st) {
    var d = by[st.ban + '|' + st.name] || { modes: {}, best: null, last: '', count: 0 };
    return { ban: st.ban, name: st.name, done: d.best !== null, best: d.best, modes: d.modes, last: d.last, count: d.count };
  });
  var top = Object.keys(miss).map(function (k) { return { item: k, n: miss[k] }; })
    .sort(function (a, b) { return b.n - a.n; }).slice(0, 10);
  return { students: out, missed: top };
}
